{%- comment -%} Sets `hit` to true when include.text matches any word (case-insensitive) or acronym (case-sensitive) of include.mod, unless it contains one of the mod's exclude_papers (case-sensitive). Callers read `hit` afterwards. {%- endcomment -%} {%- assign hit = false -%} {%- assign _low = include.text | downcase -%} {%- for w in include.mod.words -%}{%- if _low contains w -%}{%- assign hit = true -%}{%- break -%}{%- endif -%}{%- endfor -%} {%- unless hit -%}{%- for a in include.mod.acronyms -%}{%- if include.text contains a -%}{%- assign hit = true -%}{%- break -%}{%- endif -%}{%- endfor -%}{%- endunless -%} {%- if hit -%}{%- for x in include.mod.exclude_papers -%}{%- if include.text contains x -%}{%- assign hit = false -%}{%- break -%}{%- endif -%}{%- endfor -%}{%- endif -%}
