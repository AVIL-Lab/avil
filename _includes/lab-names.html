{%- comment -%} Every spelling of a lab member's name, for bolding authors (core.js). With people=true it also writes #lab-people: {slug: {name, names}} per member and alumnus, for the publications page's ?author= filter (pubs.js). The slug is `name | default: name_ko | slugify`, as in members.html. {%- endcomment -%} {%- assign names = site.data.professor.aliases -%} {%- assign everyone = site.data.members | concat: site.data.alumni -%} {%- capture people -%}{ {%- for m in everyone -%} {%- assign own = "" | split: "" -%} {%- if m.name -%}{%- assign own = own | push: m.name -%}{%- endif -%} {%- if m.name_ko -%}{%- assign own = own | push: m.name_ko -%}{%- endif -%} {%- for a in m.aliases -%}{%- assign own = own | push: a -%}{%- endfor -%} {%- assign names = names | concat: own -%} {%- assign label = m.name | default: m.name_ko -%} {{ label | slugify | jsonify }}:{"name":{{ label | jsonify }},"names":{{ own | uniq | jsonify }}}{% unless forloop.last %},{% endunless %} {%- endfor -%} }{%- endcapture -%} {%- if include.people %} {%- endif %}
